cmake_minimum_required(VERSION 3.16)
project(exige_scanner_core LANGUAGES CXX)

add_library(scanner_core STATIC src/scanner_core.cpp)
target_include_directories(scanner_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_features(scanner_core PUBLIC cxx_std_17)
set_target_properties(scanner_core PROPERTIES POSITION_INDEPENDENT_CODE ON)

if(MSVC)
  target_compile_options(scanner_core PRIVATE /W4)
else()
  target_compile_options(scanner_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
  add_executable(scanner_core_tests tests/scanner_core_test.cpp)
  target_link_libraries(scanner_core_tests PRIVATE scanner_core)
  target_compile_features(scanner_core_tests PRIVATE cxx_std_17)
  if(NOT MSVC)
    target_compile_options(scanner_core_tests PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_test(NAME scanner_core_tests COMMAND scanner_core_tests)
endif()
